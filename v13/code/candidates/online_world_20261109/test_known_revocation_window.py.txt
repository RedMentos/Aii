import time
from types import SimpleNamespace
from unittest.mock import Mock
from v13.runtime import Runtime
from v13 import world_model

def turn(r,text):
    r.submit(text)
    until=time.monotonic()+10
    while not r.idle() and time.monotonic()<until:time.sleep(.005)
    assert r.idle() and r.last_answer()['status']=='FINAL'
    return r.last_answer()['event_id']

def test_old_revocation_outside_short_window_is_not_revalidated(tmp_path):
    r=Runtime(tmp_path)
    try:
        r.sleep.cycle.set_policy('enabled','off')
        r._route=Mock(return_value=('simple',{'source':'TEST_ROUTE_SEAM'}))
        r.pool.get=Mock(return_value=SimpleNamespace(role='C3_LANGUAGE',last_metrics={},generate=lambda *a,**k:iter(['synthetic answer'])))
        r.learner.output_delta=Mock(return_value=None)
        old=turn(r,'First synthetic request');r.session=r.store.new_session();turn(r,'Second synthetic request')
        assert len(world_model.transitions(r.store))==2
        r.cfg['learning_mode']='SHADOW_UPDATE'
        result=r.online_world_update(limit=64,authority='EXPLICIT_HUMAN');assert result['learned']==2
        sid=r.store.one('SELECT source_id FROM events WHERE id=?',(old,))['source_id']
        with r.store.tx() as c:c.execute('UPDATE sources SET deleted=1 WHERE id=?',(sid,))
        status=r.online_world_update(limit=1,authority='EXPLICIT_HUMAN')
        assert status['learned']==0 and r._online_world.model.seen==2
        assert len(r._online_world.predict([0.,0.,0.,1.],'simple'))==3
    finally:r.close()
